import json
import sys
from pathlib import Path

# Ensure root project dir is in sys.path so mappings import works
SCRIPT_DIR = Path(__file__).resolve().parent
if str(SCRIPT_DIR) not in sys.path:
    sys.path.insert(0, str(SCRIPT_DIR))

from mappings.mapper import debug_mapping_coverage


def run_test(
    payload_file_path: str = "latest_tag_payload.json",
    mapping_file_path: str = "mappings/opentagv2003_to_openprinttagv1_2.json",
):
    payload_path = Path(payload_file_path)
    mapping_path = Path(mapping_file_path)

    # 1. Validate input payload path
    if not payload_path.exists():
        print(f"[Error] Payload file not found at: {payload_path.resolve()}")
        sys.exit(1)

    # 2. Validate mapping config path
    if not mapping_path.exists():
        print(f"[Error] Mapping file not found at: {mapping_path.resolve()}")
        sys.exit(1)

    # 3. Load payload and mapping configuration
    try:
        with open(payload_path, "r", encoding="utf-8") as f:
            payload = json.load(f)

        with open(mapping_path, "r", encoding="utf-8") as f:
            mapping_config = json.load(f)
    except Exception as e:
        print(f"[Error] Failed to read JSON input files: {e}")
        sys.exit(1)

    # 4. Generate and display the diagnostic report
    print("\n" + "=" * 60)
    print(" FIELD MAPPING COVERAGE REPORT")
    print("=" * 60)

    report = debug_mapping_coverage(payload, mapping_config)
    print(report)
    print("=" * 60 + "\n")


if __name__ == "__main__":
    # Allows passing custom payload/mapping paths via CLI arguments if desired
    # Usage: python run_mapping_test.py [payload.json] [mapping.json]
    payload_file = sys.argv[1] if len(sys.argv) > 1 else "latest_tag_payload.json"
    mapping_file = (
        sys.argv[2]
        if len(sys.argv) > 2
        else "mappings/opentagv2003_to_openprinttagv1_2.json"
    )

    run_test(payload_file, mapping_file)
